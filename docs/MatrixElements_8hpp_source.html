20/*!
21 @brief Result of a single matrix element calculation: the lowest-order
22 value and each correction, separately.
23 @details
24 Holds \f$ \redmatel{a}{h}{b} \f$ and its corrections,
25 \f[
26 t^{\rm tot}_{ab} = t^{(0)}_{ab} + \delta V_{ab} + \delta t^{\rm SR}_{ab}
27 + \delta t^{\rm Norm}_{ab}.
28 \f]
29 Everything is stored unscaled; the
30 accessors apply the factor that converts the reduced matrix element to the
31 requested form (reduced, stretched, or hyperfine constant); see
32 DiracOperator::MatrixElementType.
33*/
34struct MEdata {
35 //! State labels (shortSymbol); a is the bra: <a||h||b>
36 std::string a{}, b{};
37 //! Transition frequency, e_a - e_b (0 for diagonal)
38 double omega{0.0};
39 //! Factor converting reduced ME to requested MatrixElementType (1 for Reduced)
40 double factor{1.0};
41 //! Lowest-order reduced matrix element <a||h||b>
42 double t0{0.0};
43 //! RPA correction <a||dV||b> (0 if no RPA)
44 double dv{0.0};
45 //! True if an RPA correction was calculated (distinguishes from dv = 0)
46 bool has_rpa{false};
47 //! Structure radiation: top + bottom + centre diagrams (0 if no SR)
48 double sr{0.0};
49 //! Normalisation of states factor, StructureRad::norm_factor (0 if no SR)
50 double norm_factor{0.0};
51 //! Normalisation of states: norm_factor * (t0 + dv) (0 if no SR)
52 double norm{0.0};
53
54 //! Lowest-order value: factor * t0
55 double value0() const { return factor * t0; }
56 //! Lowest-order + RPA: factor * (t0 + dv)
57 double value() const { return factor * (t0 + dv); }
58 //! Full corrected value: factor * (t0 + dv + sr + norm)
59 double total() const { return factor * (t0 + dv + sr + norm); }
60};

62/*!
63 @brief Which frequency the operator, or the RPA, is evaluated at.
64 @details
65 There is no obvious default: the right choice depends on the calculation,
66 so it must be given explicitly.
67
68 - `transition`: the driver sets the frequency itself, to each pair's own
69 transition frequency \f$ \omega_{ab} = \en_a - \en_b \f$; see
70 @ref solve_at_frequency. This is the physically correct frequency
71 for a transition.
72
73 - `fixed`: the driver does not touch the frequency. The operator is assumed
74 to already be at the intended frequency, and the RPA to already have been
75 solved there, by the caller (the structure radiation tables are then
76 built once, by the driver). Use
77 for a fixed external field, or to reproduce a fixed-frequency
78 calculation.
79*/
80enum class Frequency { transition, fixed };
81
82/*!
83 @brief Options for the matrix element drivers.
84 @details
85 The two frequency choices are independent and have no default: see
86 @ref Frequency. The operator frequency usually matters most; the RPA (and
87 structure radiation) frequency dependence is often very small, so `fixed`
88 is usually adequate there.
89*/
90struct MEoptions {
91 //! Frequency of the operator itself; see Frequency
92 Frequency operator_omega;
93 //! Frequency the RPA (and the structure radiation tables) are solved at;
94 //! see Frequency. With transition and structure radiation, the SR tables
95 //! are re-built at each transition frequency, which is slow
96 Frequency rpa_omega;
97 //! Calculate diagonal matrix elements (only for even-parity operators)
98 bool diagonal{true};
99 //! Calculate off-diagonal matrix elements
100 bool off_diagonal{true};
101 //! Calculate both <a||h||b> and <b||h||a>
102 bool calculate_both{false};
103 //! Form of matrix element: Reduced, Stretched, or HFConstant
104 DiracOperator::MatrixElementType type{
105 DiracOperator::MatrixElementType::Reduced};
106 //! Maximum RPA iterations (1 corresponds to first-order RPA)
107 int rpa_iterations{128};
108 //! Print RPA solve progress
109 bool print{true};
110
111 //! The frequency choices must be made explicitly; there is no default
112 MEoptions(Frequency t_operator_omega, Frequency t_rpa_omega)
113 : operator_omega(t_operator_omega), rpa_omega(t_rpa_omega) {}
114};

116/*!
117 @brief Sets the operator frequency, solves the RPA, and re-builds the
118 structure radiation tables, for a transition at frequency omega, as the
119 options ask. The one place the frequencies are handled.
120 @details
121 In this order:
122 - with operator_omega = transition: the operator is set to
123 \f$ +|\omega| \f$, since the RPA must be solved with the operator at
124 positive frequency;
125 - with rpa_omega = transition: the RPA is re-solved at @p omega (cleared
126 first if the previous solution is poorly converged, or if first-order
127 RPA was asked for, which must not iterate from a previous solution);
128 - with operator_omega = transition: the operator is then set to the
129 signed @p omega, its physical frequency for this transition;
130 - with structure radiation, and either of the above: the SR tables are
131 re-built, so that their internal lines carry the operator and the RPA
132 as just set.
133
134 With both frequencies fixed, nothing is touched: the caller has set the
135 operator, solved the RPA, and built the SR tables.
136
137 @param h The operator (frequency updated if frequency-dependent).
138 @param dV RPA; nullptr for none.
139 @param sr Structure radiation; nullptr for none.
140 @param omega The transition frequency, signed: e_a - e_b for <a||h||b>.
141 @param options See MEoptions (the two Frequency choices, rpa_iterations,
142 print).
143*/
144void solve_at_frequency(DiracOperator::TensorOperator *h,
145 ExternalField::CorePolarisation *dV,
146 MBPT::StructureRad *sr, double omega,
147 const MEoptions &options);

149/*!
150 @brief Single matrix element of h between states a and b, with its RPA and
151 structure radiation corrections.
152 @details
153 Pure evaluation: assumes @p h is at the signed transition frequency, and
154 @p dV and @p sr solved there (see @ref solve_at_frequency). The RPA
155 correction is the component at the transition frequency: the emission one
156 (\f$ \delta V^\dagger \f$) if @p omega is negative; see
157 CorePolarisation::dV. The structure radiation denominators are taken at
158 @p omega_sr: its frequency dependence is very weak, so when the RPA is at
159 a fixed frequency this is usually the frequency the RPA was solved at
160 (see @ref sr_frequency). The Brueckner orbital correction
161 (MBPT::StructureRad::BO) is not included here.
162
163 The MatrixElementType factor is calculated with @p h (it is purely
164 angular), and stored in the returned MEdata rather than applied.
165
166 @param a,b States: <a||h||b>.
167 @param h The tensor operator.
168 @param dV RPA correction, already solved; nullptr for none.
169 @param sr Structure radiation, tables already built; nullptr for none.
170 @param type Form of matrix element (Reduced, Stretched, HFConstant).
171 @param omega Transition frequency, e_a - e_b: selects the RPA component,
172 and is recorded in the output.
173 @param omega_sr Frequency for the structure radiation denominators.
174 @return MEdata holding t0, dv, sr, norm, factor, omega, labels.
175*/
176[[nodiscard]] MEdata
177matrix_element(const DiracSpinor &a, const DiracSpinor &b,
178 const DiracOperator::TensorOperator *h,
179 const ExternalField::CorePolarisation *dV = nullptr,
180 const MBPT::StructureRad *sr = nullptr,
181 DiracOperator::MatrixElementType type =
182 DiracOperator::MatrixElementType::Reduced,
183 double omega = 0.0, double omega_sr = 0.0);
184
185/*!
186 @brief The frequency for the structure radiation denominators of a
187 transition at omega: the transition frequency if the RPA follows each
188 transition, otherwise the frequency the RPA was solved at (zero if there

317/*!
318 @brief Builds a table of reduced matrix elements, including structure
319 radiation and (optionally) the normalisation of states.
320 @details
321 As above, but each element also carries the second-order corrections,
322 \f[ t_{ab} = \redmatel{a}{h}{b} + \delta V_{ab} + \delta_{\rm SR}^{ab}, \f]
323 and every element is at the one frequency @p omega: the RPA correction is
324 \f$ \delta V(\omega) \f$ throughout (the emission component,
325 \f$ \delta V^\dagger \f$, for negative @p omega; see
326 CorePolarisation::dV), and the structure radiation is at @p omega,
327 absorbed by the ket. The reversed pair \f$ (b, a) \f$ is then
328 not the symmetry conjugate of \f$ (a, b) \f$, and is calculated as well
329 (except at \f$ \omega = 0 \f$, where it is).
330
331 @param a_orbs Bra states.
332 @param b_orbs Ket states.
333 @param h Pointer to the (const) tensor operator.
334 @param dV Optional RPA correction. If nullptr, not applied.
335 @param srn Structure radiation/normalisation. If nullptr, not applied
336 (and the table is as the plain overload above).
337 @param omega Frequency of the table: that of the operator component the
338 table stands for. Sets the sign of the RPA correction, and
339 the structure radiation denominators.
340 @param sr_n_max SR+N is applied only to pairs with both n <= @p sr_n_max.
341 SR+N is meaningful only between physical states, so this
342 limits it to the low-n part of a large basis, where the
343 states are not cavity states. Does not affect the internal
344 lines of the diagrams (see MBPT::StructureRad) [999].
345 @param sr_norm If false, only the structure radiation is added, not the
346 normalisation of states [true].
347
348 @return meTable containing t_ab for all non-zero pairs, both orderings.
349*/
350[[nodiscard]] Coulomb::meTable<double>
351me_table(const std::vector<DiracSpinor> &a_orbs,
352 const std::vector<DiracSpinor> &b_orbs,
353 const DiracOperator::TensorOperator *h,
354 const ExternalField::CorePolarisation *dV,
355 const MBPT::StructureRad *srn, double omega, int sr_n_max = 999,
356 bool sr_norm = true);
