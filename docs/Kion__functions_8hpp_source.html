54//------------------------------------------------------------------------------
55/*!
56 @brief Method used to solve bound/continuum states for form factors
57
58 @details
59 - HF : Real (Hartree-Fock) bound and continuum states (standard
60 method).
61 - Zeff : H-like (Zeff) bound and continuum states, solved
62 numerically with DiracODE.
63 - ZeffAnalytic : H-like (Zeff) bound and continuum states, using exact
64 analytic Dirac-Coulomb functions. Relativistic continuum
65 requires FLINT (see DiracContinuum::available).
66 - RPA : Hartree-Fock states, with core-polarisation (RPA)
67 corrections to every amplitude, from the outgoing-wave
68 TDHF. A method for the amplitudes, not the states: the
69 states are those of HF (see calculate_formFactors_RPA).
70*/
71enum class AtomicMethod { HF, Zeff, ZeffAnalytic, RPA };

87//------------------------------------------------------------------------------
88/*!
89 @brief Effective charge of an orbital, from its binding energy.
90
91 @details
92 \f[ Z_{\rm eff} = n\sqrt{-2\en} \f]
93 Same Zeff as used by DarkARC (see arXiv:1912.08204).
94
95 @param en Orbital energy (binding energy, negative), in au.
96 @param n Principal quantum number.
97 @return Effective charge.
98*/
99inline double Zeff_nonrel(double en, int n) {
100 return n * std::sqrt(std::abs(2.0 * en));
101}

148//------------------------------------------------------------------------------
149//! Options for the RPA (core polarisation) form factors
150struct RPAOptions {
151 //! Maximum RPA iterations per solve; 1 gives the first-order correction
152 int max_its{60};
153 //! RPA convergence target
154 double eps{1.0e-10};
155 //! An RPA solve whose final eps is above this (or nan) is discarded: the
156 //! bare (no-RPA) amplitude is used for that (E, q, K, operator)
157 double eps_fail{1.0e-3};
158 //! The RPA is solved only for energy transfers E up to this (au); above
159 //! it the factors are the bare ones (see calculate_formFactors_RPA)
160 double E_max{1.0e99};
161 //! The RPA is solved only for momentum transfers q up to this (au); in the
162 //! diagonal (q = E/c) case this is a limit on E
163 double q_max{1.0e99};
164 //! The RPA is solved only for multipoles of rank K up to this; the higher
165 //! multipoles contribute their bare factors
166 int K_max{999};
167};

169//------------------------------------------------------------------------------
170//! Bare and RPA form factors of every core orbital, from
171//! calculate_formFactors_RPA()
172struct FormFactorsRPA {
173 //! Bare (Hartree-Fock) factors of each core orbital, indexed as the core
174 std::vector<FormFactorSet> bare{};
175 //! Factors including RPA (the total, not the correction), indexed as core
176 std::vector<FormFactorSet> rpa{};
177 //! Worst RPA eps over K and operators at each (E, q); zero where no
178 //! orbital is ionised, or where the RPA was not solved (outside the
179 //! RPAOptions limits)
180 LinAlg::Matrix<double> eps{};
181 //! Number of RPA solves made (one per (E, K, operator, q) with a non-zero
182 //! amplitude, within the limits)
183 std::size_t n_solves{0};
184 //! How many of those did not converge (eps above RPAOptions::eps_fail, or
185 //! nan); zero for a first-order solve, whose eps is not tested
186 std::size_t n_failed{0};
187 //! How many of the failed solves had a converged neighbour (same K and
188 //! operator, adjacent E or q) to interpolate the RPA shift from
189 std::size_t n_interpolated{0};
190};

320//------------------------------------------------------------------------------
321/*!
322 @brief Calculates all 13 form factors (V, A, S, P) for every core orbital.
323
324 @details
325 The continuum states of each orbital are solved at every E for which the
326 ejected electron energy \f$ \en_c = E + \en_a \f$ lies in
327 (@p ec_min, @p ec_max]. The continuum l are those reached from the orbital
328 by multipoles of rank up to @p Kmax (both parities),
329 \f$ j_e = j_a \pm K_{\rm max} \f$, \f$ l_e = j_e \pm 1/2 \f$, clipped to
330 @p lc_minmax if given. Parallel over the (orbital, E) pairs.
331
332 The bound state used in the matrix elements of each orbital is
333 @p bound_states[ia]: the Hartree-Fock orbital itself, or (for the H-like
334 methods) its Zeff version, from model_bound_states(). With @p method
335 Zeff or ZeffAnalytic the continuum is also H-like, in the pointlike
336 potential of the same charge as that model state (Zeff_nonrel() of the HF
337 binding energy), solved numerically (DiracODE) or with the exact
338 analytic Dirac-Coulomb functions. The
339 occupation x_i is that of the bound state used; continuum energies always
340 use the real (HF) orbital.
341
342 @param vHF Hartree-Fock potential; its core defines the orbitals.
343 @param bound_states Bound state of each core orbital as used in the
344 matrix elements, indexed as the core.
345 @param lc_minmax Optional limits on the continuum orbital l.
346 @param ec_min Minimum ejected electron energy, in au.
347 @param ec_max Maximum ejected electron energy, in au.
348 @param force_rescale Rescale V(r) at large r for the continuum states.
349 @param hole_particle Solve the continuum in the V^(N-1) potential of the
350 hole (include the hole-particle interaction).
351 @param force_orthog Orthogonalise the continuum states to the core.
352 @param Egrid Energy transfer grid, in au.
353 @param qgrid Momentum transfer grid, in au (size 1 if diagonal).
354 @param diagonal_Eq Momentum transfer set equal to the energy transfer
355 (absorption of a massless particle).
356 @param low_q Use the low-q form of the operators.
357 @param jK_tab Precomputed spherical Bessel table.
358 @param Kmin Minimum multipolarity K.
359 @param Kmax Maximum multipolarity K.
360 @param vectorQ Calculate the vector factors.
361 @param axialQ Calculate the axial-vector factors.
362 @param scalarQ Calculate the scalar factor.
363 @param pseudoscalarQ Calculate the pseudoscalar factor.
364 @param spatialQ Calculate the spatial (E, M, L) components.
365 @param method Continuum states:
366 - AtomicMethod::HF : Hartree-Fock
367 - AtomicMethod::Zeff : H-like, DiracODE
368 - AtomicMethod::ZeffAnalytic : H-like, analytic
369 (AtomicMethod::RPA is not a states method, and is
370 treated as HF here; see calculate_formFactors_RPA.)
371 @return One FormFactorSet per core orbital, indexed as the core (zero for
372 an orbital that is not ionised anywhere on the E grid). See
373 allocate_formFactors() for which factors are calculated.
374
375 @note @p force_rescale and @p hole_particle have no effect for Zeff states.
376*/
377std::vector<FormFactorSet> calculate_formFactors(
378 const HF::HartreeFock *vHF, const std::vector<DiracSpinor> &bound_states,
379 const std::optional<std::array<int, 2>> &lc_minmax, double ec_min,
380 double ec_max, bool force_rescale, bool hole_particle, bool force_orthog,
381 const std::vector<double> &Egrid, const std::vector<double> &qgrid,
382 bool diagonal_Eq, bool low_q, const SphericalBessel::JL_table &jK_tab,
383 int Kmin, int Kmax, bool vectorQ, bool axialQ, bool scalarQ,
384 bool pseudoscalarQ, bool spatialQ, AtomicMethod method = AtomicMethod::HF);

486//------------------------------------------------------------------------------
487/*!
488 @brief One RPA solve: the bare and RPA amplitudes of every channel, for one
489 operator at one (E, K, q).
490
491 @details
492 @p h is the operator @p rpa was constructed with, with its rank and
493 frequency already set. Solves the TDHF at @p omega (warm starting from the
494 solver's current state), then fills amplitude @p i_op of each channel:
495 bare \f$ \redmatel{e}{h}{a} \f$ in @p A_bare, and RPA
496 \f$ \redmatel{e}{h + \delta V}{a} \f$ (ExternalField::TDHFcntm::dV_complex)
497 in @p A_rpa. Channels for which the operator is zero by selection rules are
498 left as they are.
499
500 If the solve did not converge (eps above RPAOptions::eps_fail, or nan), the
501 RPA amplitude is the bare one, and the solver is cleared so that the next
502 solve does not warm start from the failed state. Convergence is not tested
503 for a first-order solve (RPAOptions::max_its of 1): its eps is the size of
504 the correction, not a convergence measure. An operator with no non-zero
505 bare amplitude in any channel (a transverse multipole at K = 0) is not
506 solved: the RPA amplitudes are the bare ones, and nothing is returned.
507

509 @param i_op Its index in the multipole_operators() set.
510 @param rpa TDHF solver for @p h.
511 @param omega Energy transfer, in au.
512 @param rpa_options Iterations and convergence limits (see RPAOptions).
513 @param channels The (hole, ejected) channels (construct_channels()).
514 @param A_bare Bare amplitudes, one per channel; entry @p i_op set.
515 @param A_rpa RPA amplitudes, one per channel; entry @p i_op set.
516 @param print Print the RPA iterations of the solve.
517 @return The eps of the solve (see ExternalField::TDHF::last_eps); nullopt
518 if the RPA was not solved.
519*/
520std::optional<double> solve_channel_amplitudes(
521 const DiracOperator::TensorOperator &h, std::size_t i_op,
522 ExternalField::TDHFcntm *rpa, double omega, const RPAOptions &rpa_options,
523 const std::vector<IonisationChannel> &channels,
524 std::vector<ChannelAmplitudes> *A_bare, std::vector<ChannelAmplitudes> *A_rpa,
525 bool print = false);

527//------------------------------------------------------------------------------
528/*!
529 @brief Bare and RPA (core polarisation) form factors of every core orbital,
530 with the RPA solved at every point of the grids, from the outgoing-wave
531 TDHF.
532
533 @details
534 The bare factors are exactly those of calculate_formFactors() (Hartree-Fock
535 states only). For the RPA, each channel amplitude
536 \f$ \redmatel{e}{h}{a} \f$ is replaced by the complex outgoing-wave
537 amplitude \f$ \redmatel{e}{h + \delta V}{a} \f$ (see
538 ExternalField::TDHFcntm::dV_complex), and the factors are accumulated as in
539 accumulate_formFactors().
540
541 One RPA solve is required per (E, K, operator, q); it serves every core
542 orbital at once, which is why the orbital loop is inside. The RPA includes
543 every open channel; the ec limits apply to the output only.
544
545 Parallel over the coarsest axis that keeps every thread busy. Over the
546 energies at which some orbital is ionised, when there are at least as many
547 as threads: each thread solves its energies in full, with its own
548 continuum states and solver, and the q points in one warm-start chain.
549 Otherwise, within each (E, K, operator) block, over q when there are at
550 least as many as threads (each thread owns a solver, and its solves warm
551 start from the neighbouring q); else q runs serially with the solver's own
552 parallelism. Prints one summary line at the start, then a progress bar.
553
554 The RPA is solved at every (E, q) given: the E and q limits of
555 @p rpa_options are not applied here (see calculate_formFactors_RPA(), which
556 restricts the grids to the region within them; what the formFactors module
557 uses). The K limit is: multipoles above RPAOptions::K_max contribute their
558 bare amplitudes, with no solve.
559
560 @param vHF Hartree-Fock potential; its core defines the orbitals.
561 @param lc_minmax Optional limits on the continuum orbital l.
562 @param ec_min Minimum ejected electron energy, in au.
563 @param ec_max Maximum ejected electron energy, in au.
564 @param force_rescale Rescale V(r) at large r for the continuum states.
565 @param hole_particle Solve the continuum in the V^(N-1) potential of the
566 hole (required here; see warning).
567 @param force_orthog Orthogonalise the continuum states to the core.
568 @param Egrid Energy transfer grid, in au.
569 @param qgrid Momentum transfer grid, in au (size 1 if diagonal).
570 @param diagonal_Eq Momentum transfer set equal to the energy transfer.
571 @param low_q Use the low-q form of the operators.

706//------------------------------------------------------------------------------
707/*!
708 @brief Corrects the points at which the RPA solve failed, by interpolating
709 the relative RPA shift from the neighbouring columns.
710
711 @details
712 Where a solve fails the bare value is used, which leaves a step in an
713 otherwise smooth function. The relative shift \f$ R = K_{\rm RPA}/K_{\rm
714 bare} \f$ varies slowly along the columns (q for the form factors, the
715 photon energy for photoionisation), so it is taken from the converged
716 neighbours and applied to the bare value,
717 \f$ K_{\rm RPA}(i,j) = R\,K_{\rm bare}(i,j) \f$:
718 \f[
719 \begin{align}
720 R &= \frac{1}{2}(R_{j-1} + R_{j+1}), \\
721 R &= 1 + \frac{1}{2}(R_{j\mp1} - 1),
722 \end{align}
723 \f]
724 the mean of the two when both have converged; half the relative correction
725 when only one has, since the shift is then unconstrained on the other side.
726 A point with no converged neighbour (in a run of adjacent failures, a
727 resonance say) is left as it is, as is one whose converged neighbours have
728 a zero bare value. Points that fail are never used as neighbours, so the
729 result does not depend on the order of the corrections.
730
731 @param eps Final RPA eps of each point (rows x columns).
732 @param eps_fail Failure threshold (see rpa_failed()).
733 @param K_bare Bare (no-RPA) values, same shape as @p eps.
734 @param K_rpa RPA values, same shape; corrected in place.
735
736 @note Corrects nothing with a single column, since there is no neighbour
737 to interpolate from. Only meaningful for an iterated solve: after a
738 first-order solve (RPAOptions::max_its of 1) eps is the size of the
739 correction, not a convergence measure.
740*/
741void interpolate_failed_rpa(const LinAlg::Matrix<double> &eps, double eps_fail,
742 const LinAlg::Matrix<double> &K_bare,
743 LinAlg::Matrix<double> *K_rpa);

749
750 @details
