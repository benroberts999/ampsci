148/*!
149 @brief Nuclear potential: uniformly charged sphere (ball) of radius rnuc.
150
151 @details
152 \f[
153 V(r) = \begin{cases}
154 -\frac{Z}{2 r_N}\left(3 - r^2/r_N^2\right) & r < r_N, \\
155 -Z/r & r \geq r_N.
156 \end{cases}
157 \f]
158 rnuc = 0 gives a point-like nucleus.
159
160 @param Z Nuclear charge
161 @param rnuc Nuclear charge radius, in fm (not r_rms: for a uniform
162 sphere, r_N = sqrt(5/3) r_rms)
163 @param rgrid Radial points (atomic units)
164
165 @return V(r) at each point of @p rgrid (atomic units).
166*/
167[[nodiscard]] std::vector<double>
168sphericalNuclearPotential(double Z, double rnuc,
169 const std::vector<double> &rgrid);
170
171/*!
172 @brief Nuclear potential: Gaussian charge distribution.
173
174 @details
175 \f[
176 V(r) = -\frac{Z}{r}\,{\rm erf}\left(\sqrt{3/2}\;r/r_{\rm rms}\right).
177 \f]
178
179 @param Z Nuclear charge
180 @param r_rms Root-mean-square charge radius, in fm
181 @param rgrid Radial points (atomic units)
182
183 @return V(r) at each point of @p rgrid (atomic units).
184*/
185[[nodiscard]] std::vector<double>
186GaussianNuclearPotential(double Z, double r_rms,
187 const std::vector<double> &rgrid);
188
189/*!
190 @brief Nuclear potential: Fermi charge distribution.
191
192 @details
193 For charge density
194 \f[
195 \rho(r) = \rho_0 \left[ 1 + e^{(r - c)/a} \right]^{-1},
196 \qquad a = t/(4\ln 3),
197 \f]
198 normalised to Z. Expressed in terms of complete Fermi-Dirac integrals
199 (GSL). V = -Z/r beyond r = 10c.
200
201 @param Z Nuclear charge
202 @param t Skin thickness (90% to 10% fall-off), in fm
203 @param c Half-density radius, in fm (not r_rms)
204 @param rgrid Radial points (atomic units)
205
206 @return V(r) at each point of @p rgrid (atomic units).
207*/
208[[nodiscard]] std::vector<double>
209fermiNuclearPotential(double Z, double t, double c,
210 const std::vector<double> &rgrid);
211
212/*!
213 @brief Fermi charge distribution, rho(r), normalised to Z_norm.
214
215 @details
216 \f$ \int \rho\, d^3r = 4\pi \int \rho(r)\, r^2 dr = Z_{\rm norm} \f$:
217 Z_norm = Z for nuclear charge density; Z_norm = 1 for nuclear density.
218 rho = 0 beyond r = 10c.
219
220 @param t Skin thickness, in fm
221 @param c Half-density radius, in fm
222 @param Z_norm Normalisation
223 @param grid Radial grid
224
225 @return rho(r) on @p grid (atomic units).
226*/
227[[nodiscard]] std::vector<double>
228fermiNuclearDensity_tcN(double t, double c, double Z_norm, const Grid &grid);

230/*!
231 @brief Nuclear potential V(r) for given nucleus, at radial points @p r.
232
233 @details
234 Calls one of the above, depending on nucleus type. If r_rms = 0, point-like
235 (-Z/r) for all types except custom.
236 - Fermi: c from r_rms and t; if beta is non-zero, uses the effective skin
237 thickness for the deformed nucleus (deformation_effective_t)
238 - spherical: radius r_N = sqrt(5/3) r_rms
239 - custom: reads {r, V(r)} (atomic units) from custom_pot_file(), and
240 interpolates onto @p r. Beyond the last point, V = -Z_eff/r, with Z_eff
241 from the last point. Not extrapolated to smaller r.
242
243 @return V(r) at each point of @p r (atomic units).
244
245 @warning Aborts if the custom file is invalid (fewer than 3 points), or the
246 type is invalid.
247*/
248[[nodiscard]] std::vector<double> formPotential(const Nucleus &nucleus,
249 const std::vector<double> &r);
