14enum class Type { Green, Tietz };

16/*!
17 @brief Green parametric potential at r; electron part only (no nuclear).
18
19 @details
20 \f[
21 V(r) = \frac{Z}{r} - \frac{1}{r}
22 \left[ 1 + \frac{Z-1}{H\,(e^{r/d} - 1) + 1} \right].
23 \f]
24
25 The total potential is V(r) - Z/r.
26 Returns 0 at r = 0, and for Z = 1 (no screening electrons).
27*/
28double green(int Z, double r, double H, double d);
29
30/*!
31 @brief Tietz parametric potential at r; electron part only (no nuclear).
32
33 @details
34 \f[
35 V(r) = \frac{Z}{r} - \frac{1}{r}
36 \left[ 1 + \frac{(Z-1)\,e^{-g r}}{(1 + t r)^2} \right].
37 \f]
38 The total potential is V(r) - Z/r. Returns 0 at r = 0.
39*/
40double tietz(int Z, double r, double g, double t);
41
42/*!
43 @brief Default Green parameters H and d for given Z, tuned to match HF core.
44
45 @details
46 Tabulated for selected Z, linearly interpolated in between. For Z > 113,
47 H = 5.0, d = 0.75.
48
49 @return Always 0; H and d are set by reference.
50
51 @note For Z = 1, H = d = 0: unused, since there is no screening.
52*/
53int defaultGreenCore(int z, double &H, double &d);
54
55/*!
56 @brief Default Green parameters H and d for given Z, tuned to experiment.
57
58 @details
59 Fitted to the lowest valence J=1/2 states for selected Z; crude quadratic
60 fit in Z otherwise.
61
62 @return Always 0; H and d are set by reference.
63*/
64int defaultGreen(int z, double &H, double &d);
65
66/*!
67 @brief Default Tietz parameters g and t for given Z, tuned to experiment.
68
69 @details
70 Fitted to the lowest valence J=1/2 states for selected Z; crude quadratic
71 fit in Z otherwise.
72
73 @return Always 0; g and t are set by reference.
74*/
75int defaultTietz(int z, double &g, double &t);
